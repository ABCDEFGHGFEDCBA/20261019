//建立顺序表：先输入个数 n，再输入 n 个整数 
sequenlist* create()
{
    sequenlist* L = (sequenlist*)malloc(sizeof(sequenlist));
    L->last = 0;   // 空表

    int n, x;
    printf("请输入元素个数：");
    scanf_s("%d", &n);

    if (n < 0) {
        printf("元素个数不能为负！\n");
        return L;
    }
    if (n > MAXSIZE - 1) {  // 因为 data[0] 不用，最多存 MAXSIZE-1 个
        printf("元素个数超过容量，最多 %d 个！\n", MAXSIZE - 1);
        n = MAXSIZE - 1;
    }

    printf("请输入 %d 个整数：", n);
    for (int i = 0; i < n; i++) {
        scanf_s("%d", &x);
        L->last++;
        L->data[L->last] = x;
    }
    return L;
}

/* 输出顺序表 */
void print(sequenlist* L)
{
    if (L->last == 0) {
        printf("空表\n");
        return;
    }
    for (int i = 1; i <= L->last; i++) {
        printf("%3d", L->data[i]);
    }
    printf("\n");
}

    } else {
        printf("\n ***********该链表不存在环\n");
        printf("***********无环链表显示如下：***************\n");

//尾插法建立带头结点的单链表，'*' 结束 
linklist* create()
{
    linklist *head, *s, *r;
    char ch;
    head = (linklist*)malloc(sizeof(linklist));
    r = head;

    while ((ch = getchar()) != '*') {
        s = (linklist*)malloc(sizeof(linklist));
        s->data = ch;
        r->next = s;
        r = s;
    }
    r->next = NULL;
    return head;
}

//========== 无环打印：a->b->c->NULL ========== 
void printnocycle(linklist *head)
{
    linklist *p = head->next;

    if (p == NULL) {
        printf("该表为空表\n");
        return;
    }

    while (p != NULL) {
        printf("%c->", p->data);
        p = p->next;
    }
    printf("NULL\n");
}

// ========== 有环打印：标出环入口和环 ========== 

void printcycle(linklist *head)
{
    linklist *entry = findCycleEntry(head);   // 先找环入口

    if (entry == NULL) {
        printf("该表无环，请使用 printnocycle\n");
        return;
    }

    //第一步：打印头到环入口之前的部分
    linklist *p = head->next;
    while (p != entry) {
        printf("%c->", p->data);
        p = p->next;
    }

    //第二步：打印环部分，用括号括起来
    printf("(%c", entry->data);
    p = entry->next;
    while (p != entry) {
        printf("->%c", p->data);
        p = p->next;
    }
    printf(")->...\n");
}
